import fs from "node:fs/promises"; const GEMINI_MODEL = "gemini-3.5-flash-lite"; const BATCH_SIZE = 4; const USER_AGENT = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/153 Safari/537.36"; function cleanText(value = "") { return String(value) .replace(/ /gi, " ").replace(/&/gi, "&").replace(/"/gi, '"') .replace(/'/gi, "'").replace(/'/gi, "'").replace(/’/gi, "'") .replace(/‘/gi, "'").replace(/“/gi, '"').replace(/”/gi, '"') .replace(/–/gi, "-").replace(/—/gi, "-").replace(/…/gi, "...") .replace(/&#(\d+);/g, (_, n) => String.fromCharCode(Number(n))) .replace(/https?:\/\/\S+/gi, " ").replace(/\s+/g, " ").trim(); } function stripHtml(html = "") { return cleanText(String(html) .replace(//gi, " ") .replace(//gi, " ") .replace(//gi, " ") .replace(//gi, " ") .replace(/<[^>]+>/g, " ")); } function meta(html, name) { const patterns = [ new RegExp(`]+property=["']${name}["'][^>]+content=["']([^"']+)["']`, "i"), new RegExp(`]+content=["']([^"']+)["'][^>]+property=["']${name}["']`, "i"), new RegExp(`]+name=["']${name}["'][^>]+content=["']([^"']+)["']`, "i"), new RegExp(`]+content=["']([^"']+)["'][^>]+name=["']${name}["']`, "i") ]; for (const pattern of patterns) { const match = html.match(pattern); if (match?.[1]) return cleanText(match[1]); } return ""; } function extractArticleTitle(html, fallback = "") { const h1 = html.match(/]*>([\s\S]*?)<\/h1>/i); return cleanText(h1?.[1] ? stripHtml(h1[1]) : meta(html, "og:title") || meta(html, "twitter:title") || fallback); } function extractBody(html = "") { const candidates = []; for (const match of html.matchAll(/]*>([\s\S]*?)<\/article>/gi)) candidates.push(stripHtml(match[1] || "")); for (const match of html.matchAll(/]*>([\s\S]*?)<\/main>/gi)) candidates.push(stripHtml(match[1] || "")); const paragraphs = [...html.matchAll(/]*>([\s\S]*?)<\/p>/gi)] .map(m => stripHtml(m[1] || "")) .filter(x => x.length >= 45 && !/^(advertisement|read more|subscribe|follow us|share|home|menu)$/i.test(x)); if (paragraphs.length) candidates.push(paragraphs.join(" ")); return candidates.sort((a, b) => b.length - a.length)[0]?.slice(0, 9000) || ""; } async function fetchText(url) { const response = await fetch(url, { headers: { "User-Agent": USER_AGENT, "Accept": "text/html,application/xhtml+xml" } }); if (!response.ok) throw new Error(`Source HTTP ${response.status}`); return response.text(); } function extractJson(text = "") { const cleaned = String(text).replace(/^\s*```json\s*/i, "").replace(/\s*```\s*$/i, "").trim(); try { return JSON.parse(cleaned); } catch {} const first = cleaned.indexOf("{"); const last = cleaned.lastIndexOf("}"); if (first >= 0 && last > first) { try { return JSON.parse(cleaned.slice(first, last + 1)); } catch {} } return null; } async function callGemini(prompt) { const key = process.env.GEMINI_API_KEY; if (!key) throw new Error("GEMINI_API_KEY is missing"); const response = await fetch( `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${encodeURIComponent(key)}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ systemInstruction: { parts: [{ text: "You are Cineinsta's original cinema editor. Return JSON only." }] }, contents: [{ role: "user", parts: [{ text: prompt }] }], generationConfig: { temperature: 0.2, responseMimeType: "application/json" } }) } ); if (!response.ok) { const detail = (await response.text()).slice(0, 800); throw new Error(`Gemini HTTP ${response.status}: ${detail}`); } const data = await response.json(); const text = (data?.candidates?.[0]?.content?.parts || []).map(p => p?.text || "").join(""); const parsed = extractJson(text); if (!parsed) throw new Error("Gemini returned invalid JSON"); return parsed; } function validArticle(story) { if (!story || story.keep !== true) return false; const title = cleanText(story.title); const summary = cleanText(story.summary); const body = Array.isArray(story.body) ? story.body.map(cleanText).filter(Boolean) : []; const words = body.join(" ").split(/\s+/).filter(Boolean).length; if (title.length < 25 || title.length > 140) return false; if (summary.length < 120 || summary.length > 650) return false; if (body.length < 5 || body.length > 10) return false; if (words < 250 || words > 900) return false; if (body.some(p => p.length < 45)) return false; const junk = /(read the original|read more at|source:|according to .*times|subscribe now|click here)/i; if (junk.test(body.join(" "))) return false; return true; } function buildPrompt(batch) { const stories = batch.map((s, i) => `STORY ${i + 1} SOURCE HEADLINE: ${s.sourceTitle} SOURCE ARTICLE: ${s.body}` ).join("\n\n====================\n\n"); return `You are the original editorial writer for Cineinsta, an Indian cinema news website. Rewrite ALL ${batch.length} stories below. For every kept story: - Create a fresh Cineinsta headline. - Create a 2-3 sentence homepage summary. - Create the COMPLETE article as 5-10 readable paragraphs, about 250-900 words. - Use only facts supported by the supplied article. - Rewrite in original language; do not copy sentences, paragraph order, or distinctive wording. - Do not mention the source publisher, source website, source article, or provide a source link. - Do not say "according to the source", "read more", "read the original", or similar. - Do not invent facts, quotes, dates, numbers, motives or details. - Do not pad the article with generic filler just to reach a word count. - If the source does not contain enough reliable facts for a useful article, set keep=false. Return exactly one result per story in the same order. JSON only: {"stories":[{"keep":true,"title":"...","summary":"...","category":"Telugu Cinema","body":["paragraph 1","paragraph 2","paragraph 3","paragraph 4","paragraph 5"]}]} ${stories}`; } async function prepare(item, index) { if (!item?.url) throw new Error(`No URL for story ${index + 1}`); const html = await fetchText(item.url); const sourceTitle = extractArticleTitle(html, item.title); const body = extractBody(html); if (body.length < 250) throw new Error("Insufficient source article text"); return { item, sourceTitle, body, sourceIndex: index }; } async function main() { if (!process.env.GEMINI_API_KEY) throw new Error("GEMINI_API_KEY is missing"); const feedPath = "data/feed.json"; const feed = JSON.parse(await fs.readFile(feedPath, "utf8")); if (!Array.isArray(feed.news) || !feed.news.length) throw new Error("No news stories in data/feed.json"); const prepared = []; for (let i = 0; i < feed.news.length; i++) { try { prepared.push(await prepare(feed.news[i], i)); } catch (error) { console.log(`SOURCE SKIP ${i + 1}: ${error.message}`); } } if (!prepared.length) throw new Error("No usable source stories"); const rewrittenByIndex = new Map(); for (let start = 0; start < prepared.length; start += BATCH_SIZE) { const batch = prepared.slice(start, start + BATCH_SIZE); console.log(`Gemini batch ${Math.floor(start / BATCH_SIZE) + 1}: ${batch.length} stories`); let response; try { response = await callGemini(buildPrompt(batch)); } catch (error) { console.log(`BATCH SKIP: ${error.message}`); // Do not retry individual stories: that can exhaust the Gemini free quota. continue; } if (!Array.isArray(response?.stories) || response.stories.length !== batch.length) { console.log(`BATCH SKIP: Gemini returned ${response?.stories?.length || 0} results for ${batch.length} stories`); continue; } for (let i = 0; i < batch.length; i++) { const story = response.stories[i]; if (!validArticle(story)) { console.log(`REJECTED: ${batch[i].sourceTitle}`); continue; } const body = story.body.map(cleanText).filter(Boolean); const rewritten = { ...batch[i].item, title: cleanText(story.title), summary: cleanText(story.summary), category: cleanText(story.category || batch[i].item.category || "Telugu Cinema"), body, keyFacts: [], cineinstaContext: "", editorial: "Cineinsta", // Keep these only for internal feed traceability. The site renderer must not display them. source: batch[i].item.source, url: batch[i].item.url }; rewrittenByIndex.set(batch[i].sourceIndex, rewritten); console.log(`ACCEPTED: ${rewritten.title} (${body.join(" ").split(/\s+/).length} words)`); } } // Never replace the feed with an incomplete/mostly failed run. // Existing stories are retained if they were not successfully rewritten. let changed = 0; for (const [index, rewritten] of rewrittenByIndex) { feed.news[index] = rewritten; changed++; } if (changed === 0) { throw new Error("No complete articles were generated. Existing feed was not changed."); } feed.updatedAt = new Date().toISOString(); await fs.writeFile(feedPath, JSON.stringify(feed, null, 2) + "\n", "utf8"); console.log("======================================"); console.log(`Complete Cineinsta articles updated: ${changed}`); console.log(`Total feed stories retained: ${feed.news.length}`); console.log("Gemini was called only once per batch; no individual retry storm."); } main().catch(error => { console.error("Cineinsta editorial rewrite failed:"); console.error(error); process.exit(1); });
